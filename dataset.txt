C14 - Bubble Sort
Version A (Unclear):
public void f(int[] a) {
    for(int i = 0; i < a.length-1; i++)
        for(int j = 0; j < a.length-i-1; j++)
            if(a[j] > a[j+1]) {
                int t = a[j];

                a[j+1] = t;
            }
}
Version B (Meaningful):
public void bubbleSort(int[] array) {
    for(int outerIndex = 0; outerIndex < array.length-1; outerIndex++)
        for(int innerIndex = 0; innerIndex < array.length-outerIndex-1; innerIndex++)
            if(array[innerIndex] > array[innerIndex+1]) {
                int temp = array[innerIndex];
                array[innerIndex] = array[innerIndex+1];
                array[innerIndex+1] = temp;
            }
}

C16 - Binary Search
Version A (Unclear):
public int f(int[] a, int t) {
    int l = 0, r = a.length-1;
    while(l <= r) {
        int m = (l+r)/2;
        if(a[m] == t) return m;
        if(a[m] < t) l = m+1;
        else r = m-1;
    }
    return -1;
}
Version B (Meaningful):
public int binarySearch(int[] sortedArray, int target) {
    int left = 0, right = sortedArray.length-1;
    while(left <= right) {
        int middle = (left+right)/2;
        if(sortedArray[middle] == target) return middle;
        if(sortedArray[middle] < target) left = middle+1;
        else right = middle-1;
    }
    return -1;
}

C17 - Merge Two Sorted Arrays
Version A (Unclear):
public int[] f(int[] a, int[] b) {
    int[] r = new int[a.length+b.length];
    int i = 0, j = 0, k = 0;
    while(i < a.length && j < b.length) {
        if(a[i] < b[j]) r[k++] = a[i++];
        else r[k++] = b[j++];
    }
    while(i < a.length) r[k++] = a[i++];
    while(j < b.length) r[k++] = b[j++];
    return r;
}
Version B (Meaningful):
public int[] mergeArrays(int[] firstArray, int[] secondArray) {
    int[] merged = new int[firstArray.length+secondArray.length];
    int firstIdx = 0, secondIdx = 0, mergeIdx = 0;
    while(firstIdx < firstArray.length && secondIdx < secondArray.length) {
        if(firstArray[firstIdx] < secondArray[secondIdx]) merged[mergeIdx++] = firstArray[firstIdx++];
        else merged[mergeIdx++] = secondArray[secondIdx++];
    }
    while(firstIdx < firstArray.length) merged[mergeIdx++] = firstArray[firstIdx++];
    while(secondIdx < secondArray.length) merged[mergeIdx++] = secondArray[secondIdx++];
    return merged;
}
